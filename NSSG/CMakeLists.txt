cmake_minimum_required(VERSION 3.20)
project(nssg_search_project LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on Linux or WSL.")
endif()
add_executable(nssg_search src/search.cpp vendor/ssg/src/util.cpp)
target_compile_options(nssg_search PRIVATE -O3 -pthread)
target_link_libraries(nssg_search PRIVATE pthread)
target_include_directories(nssg_search PRIVATE "${PROJECT_SOURCE_DIR}/../ann_quantization")
target_include_directories(nssg_search SYSTEM PRIVATE vendor/ssg/include vendor/deps/include)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    target_compile_options(nssg_search PRIVATE -mcpu=native)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    target_compile_options(nssg_search PRIVATE -march=native)
endif()

include(cmake/build_targets.cmake)
