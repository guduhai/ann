cmake_minimum_required(VERSION 3.20)
project(cagra_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(ANDROID)
    include(cmake/android.cmake)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(cmake/linux.cmake)
else()
    message(FATAL_ERROR "Supported targets: Linux and Android arm64-v8a")
endif()
option(ANN_SANITIZE "Enable address/undefined sanitizers" OFF)
target_compile_options(ann_platform INTERFACE -Wall -Wextra -ffp-contract=off)
if(ANN_SANITIZE)
    target_compile_options(ann_platform INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ann_platform INTERFACE -fsanitize=address,undefined)
endif()
include(cmake/build_targets.cmake)
option(BUILD_TESTING "Build the detour pruning rule test" OFF)
if(BUILD_TESTING)
    enable_testing()
    add_executable(test_prune tests/test_prune.cpp)
    target_include_directories(test_prune PRIVATE include)
    target_link_libraries(test_prune PRIVATE ann_platform)
    add_test(NAME detour_rule COMMAND test_prune)
endif()
