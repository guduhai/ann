cmake_minimum_required(VERSION 3.16)
project(standalone_cagra LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ANN_SANITIZE "Enable address/undefined sanitizers" OFF)
add_compile_options(-Wall -Wextra -ffp-contract=off)
if(ANN_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(OpenMP REQUIRED)
add_executable(ann_knn src/build_knn.cpp vendor/efanna_graph/src/index.cpp vendor/efanna_graph/src/index_graph.cpp vendor/efanna_graph/src/index_random.cpp)
target_include_directories(ann_knn PRIVATE include vendor/efanna_graph/include)
target_link_libraries(ann_knn PRIVATE OpenMP::OpenMP_CXX)
add_executable(ann_build src/build_graph.cpp vendor/nsg/src/index.cpp vendor/nsg/src/index_nsg.cpp)
target_include_directories(ann_build PRIVATE include vendor/nsg/include)
target_link_libraries(ann_build PRIVATE OpenMP::OpenMP_CXX)
add_executable(ann_query src/query.cpp)
target_include_directories(ann_query PRIVATE include)
add_executable(test_prune tests/test_prune.cpp)
target_include_directories(test_prune PRIVATE include)
enable_testing()
add_test(NAME detour_rule COMMAND test_prune)
