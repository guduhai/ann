cmake_minimum_required(VERSION 3.20)
project(hnsw_search_project LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(ANDROID)
    include(cmake/android.cmake)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(cmake/linux.cmake)
else()
    message(FATAL_ERROR "Supported targets: Linux and Android arm64-v8a")
endif()
add_executable(hnsw_search src/search.cpp)
target_compile_options(hnsw_search PRIVATE -O3)
target_link_libraries(hnsw_search PRIVATE ann_platform)
target_include_directories(hnsw_search PRIVATE "${PROJECT_SOURCE_DIR}/../ann_quantization")
target_include_directories(hnsw_search SYSTEM PRIVATE vendor/hnswlib)
include(cmake/build_targets.cmake)
add_custom_command(TARGET hnsw_search POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:hnsw_search>" "${CMAKE_BINARY_DIR}/bin/ann_query")
